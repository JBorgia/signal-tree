/** Isolated reachability falsifier. No raw history snapshots are read or retained. */
import { signalTree } from '../../../packages/kernel/src/lib/signal-tree';
import { leaf } from '../../../packages/kernel/src/lib/leaf';
import {
  transactions,
  peekInternalTransactionRuntime,
} from '../../../packages/kernel/src/enhancers/transactions/transactions';
import { confirmedTurnReader } from '../../../packages/kernel/src/internals';
import { external } from '../../../packages/kernel/src/lib/external';

type Payload = { tag: string; bytes?: Uint8Array };
const flush = async () => {
  await Promise.resolve();
  await Promise.resolve();
};
async function createCase(mode: string) {
  const tree = signalTree(
    { payload: leaf<Payload>({ tag: 'construction' }) },
    { enhancers: [transactions()] }
  );
  const runtime = peekInternalTransactionRuntime(tree)!;
  const reader = confirmedTurnReader(tree)!;
  const obsolete: Payload = {
    tag: 'obsolete',
    bytes: new Uint8Array(1024 * 1024),
  };
  const weak = new WeakRef(obsolete);
  const replacement: Payload = { tag: 'current' };
  if (mode === 'destroyed-external-held') {
    external(() => tree.$.payload(obsolete));
    await flush();
    external(() => tree.$.payload(replacement));
    await flush();
  } else if (mode === 'destroyed-rejected-held') {
    tree.transact(() => tree.$.payload(obsolete)).rollback();
    await flush();
    external(() => tree.$.payload(replacement));
    await flush();
  } else {
    tree.transact(() => tree.$.payload(obsolete)).confirm();
    await flush();
    tree.transact(() => tree.$.payload(replacement)).confirm();
    await flush();
  }
  if (tree.$.payload().tag !== 'current')
    throw new Error('Old payload is still current state: invalid probe');
  const confirmedBeforeDestroy = runtime.getConfirmedTurnCount();
  if (mode !== 'live-confirmed-held') tree.destroy();
  await flush();
  let readerRefusesAfterDestroy: boolean | null = null;
  if (tree.destroyed()) {
    try {
      reader.readConfirmedTurns();
      readerRefusesAfterDestroy = false;
    } catch (error) {
      readerRefusesAfterDestroy =
        error instanceof Error &&
        error.message.includes('STUDIO_TREE_DESTROYED');
    }
  }
  const held =
    mode === 'destroyed-confirmed-released' ? null : { tree, runtime, reader };
  return { weak, held, confirmedBeforeDestroy, readerRefusesAfterDestroy };
}
export async function gcCase(mode: string) {
  if (typeof globalThis.gc !== 'function')
    throw new Error('GC probe requires --expose-gc');
  const result = await createCase(mode);
  // WeakRef targets are kept alive for the job in which they are created or
  // dereferenced. Never dereference between collections; yield entire jobs.
  for (let i = 0; i < 16; i++) {
    await new Promise<void>((resolve) => setImmediate(resolve));
    globalThis.gc();
    await new Promise<void>((resolve) => setImmediate(resolve));
  }
  const retained = result.weak.deref() !== undefined;
  const shouldRetain = mode === 'live-confirmed-held';
  const observed = {
    mode,
    obsoletePayloadRetained: retained,
    expectedRetention: shouldRetain,
    status: retained === shouldRetain ? 'held' : 'violated',
    holdsTreeRuntimeReader: result.held !== null,
    confirmedBeforeDestroy: result.confirmedBeforeDestroy,
    confirmedAfterDestroy: result.held?.runtime.getConfirmedTurnCount() ?? null,
    readerRefusesAfterDestroy: result.readerRefusesAfterDestroy,
    destroyed: result.held?.tree.destroyed() ?? true,
    heapUsedAfterGC: process.memoryUsage().heapUsed,
    scope:
      'No raw record/snapshot read retained. Target was introduced after construction and replaced before destroy. Live-tree/current-value reads never return the obsolete payload.',
  };
  result.held?.tree.destroy();
  return observed;
}
